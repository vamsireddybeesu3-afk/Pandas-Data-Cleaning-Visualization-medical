import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import seaborn as sns
df=pd.read_csv(r"C:\Users\Vykunta Reddy\Downloads\practise for pandas ecommerce1.csv")
print(df.head())
print(df.shape)
print(df.isnull().sum())
print(df.duplicated().sum())
df=df.drop_duplicates()
df.columns=df.columns.str.strip()
text_columns=[
    "Customer Name","City","Region","Segment","Product","Category","Payment Mode"
]
for col in text_columns:
    df[col]=df[col].astype("string").str.strip()
df["Region"]=df["Region"].str.title()
df["Product"]=df["Product"].str.title()
df["Payment Mode"]=df["Payment Mode"].str.title()
df["Order Date"]=pd.to_datetime(
    df["Order Date"],errors="coerce"
)
df["Sales"]=pd.to_numeric(
    df["Sales"].astype(str)
    .str.replace(r"[$,]","",regex=True),
    errors="coerce"
)
df["Quantity"]=pd.to_numeric(
    df["Quantity"].astype(str)
    .str.replace("units","",regex=False)
    .str.strip(),
    errors="coerce"
)
df["Unit Price"] = pd.to_numeric(
    df["Unit Price"], errors="coerce"
)
df["Profit"]=pd.to_numeric(
    df["Profit"],errors="coerce"
)
df["Discount"]=pd.to_numeric(
    df["Discount"],errors="coerce"
)
print(df.info())
print(df.isnull().sum())
print(df.duplicated().sum())
print(df[["Sales","Profit","Quantity"]].describe())
